'use client';

import { useState, useEffect } from 'react';
import Link from 'next/link';

// Mock Signal Generator
const generateSignals = () => {
  const pairs = ['EUR/USD', 'GBP/JPY', 'USD/JPY', 'BTC/USD', 'ETH/USD', 'XAU/USD'];
  const types = ['BUY', 'SELL'];
  
  return pairs.map(pair => {
    const type = types[Math.floor(Math.random() * types.length)];
    const confidence = Math.floor(Math.random() * 25) + 75; // 75-99%
    const entry = (Math.random() * 100 + 1).toFixed(4);
    const sl = (parseFloat(entry) - (type === 'BUY' ? 0.005 : -0.005)).toFixed(4);
    const tp = (parseFloat(entry) + (type === 'BUY' ? 0.015 : -0.015)).toFixed(4);
    
    return { 
      pair, 
      type, 
      confidence, 
      entry, 
      sl, 
      tp,
      time: new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' })
    };
  });
};

export default function Signals() {
  const [signals, setSignals] = useState<any[]>([]);
  const [loading, setLoading] = useState(true);
  const [executed, setExecuted] = useState<string[]>([]);

  useEffect(() => {
    setTimeout(() => {
      setSignals(generateSignals());
      setLoading(false);
    }, 1000);
  }, []);

  const executeSignal = (pair: string) => {
    setExecuted([...executed, pair]);
    setTimeout(() => {
      setExecuted(executed.filter(p => p !== pair));
    }, 2000);
  };

  return (
    <div className="min-h-screen bg-gray-900 text-white p-8">
      <div className="max-w-6xl mx-auto">
        {/* Header */}
        <div className="mb-8 flex justify-between items-center">
          <div>
            <h1 className="text-3xl font-bold text-green-400 mb-2">AI Trading Signals</h1>
            <p className="text-gray-400">Real-time trade opportunities generated by Matrix AI.</p>
          </div>
          <Link href="/" className="text-gray-400 hover:text-white transition flex items-center gap-2">
            ← Back to Dashboard
          </Link>
        </div>

        {/* Stats Bar */}
        <div className="grid grid-cols-1 md:grid-cols-3 gap-4 mb-8">
          <div className="bg-gray-800 p-4 rounded-xl border border-gray-700">
            <p className="text-xs text-gray-400 uppercase tracking-wider">Active Signals</p>
            <p className="text-2xl font-bold text-white">{signals.length}</p>
          </div>
          <div className="bg-gray-800 p-4 rounded-xl border border-gray-700">
            <p className="text-xs text-gray-400 uppercase tracking-wider">Avg Confidence</p>
            <p className="text-2xl font-bold text-green-400">
              {signals.length > 0 
                ? Math.round(signals.reduce((a, b) => a + b.confidence, 0) / signals.length) 
                : 0}%
            </p>
          </div>
          <div className="bg-gray-800 p-4 rounded-xl border border-gray-700">
            <p className="text-xs text-gray-400 uppercase tracking-wider">Last Updated</p>
            <p className="text-2xl font-bold text-blue-400">
              {new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' })}
            </p>
          </div>
        </div>

        {/* Loading State */}
        {loading ? (
          <div className="space-y-4">
            {[1, 2, 3].map(i => (
              <div key={i} className="bg-gray-800 p-6 rounded-xl border border-gray-700 animate-pulse">
                <div className="h-4 bg-gray-700 rounded w-1/4 mb-4"></div>
                <div className="h-8 bg-gray-700 rounded w-1/2 mb-4"></div>
                <div className="h-4 bg-gray-700 rounded w-full"></div>
              </div>
            ))}
          </div>
        ) : (
          /* Signals List */
          <div className="space-y-4">
            {signals.map((signal, idx) => (
              <div 
                key={idx} 
                className={`bg-gray-800 p-6 rounded-xl border transition-all duration-300 ${
                  executed.includes(signal.pair) 
                    ? 'border-green-500 bg-green-900/20' 
                    : 'border-gray-700 hover:border-gray-600'
                }`}
              >
                <div className="flex flex-col md:flex-row md:items-center justify-between gap-4">
                  
                  {/* Left: Pair & Type */}
                  <div className="flex items-center gap-4">
                    <div className={`w-12 h-12 rounded-full flex items-center justify-center text-lg font-bold ${
                      signal.type === 'BUY' ? 'bg-green-900/50 text-green-400' : 'bg-red-900/50 text-red-400'
                    }`}>
                      {signal.type === 'BUY' ? '↑' : '↓'}
                    </div>
                    <div>
                      <h3 className="text-xl font-bold text-white">{signal.pair}</h3>
                      <p className="text-sm text-gray-400">Generated at {signal.time}</p>
                    </div>
                  </div>

                  {/* Middle: Levels */}
                  <div className="grid grid-cols-3 gap-4 flex-1">
                    <div className="bg-gray-900/50 p-3 rounded-lg">
                      <p className="text-xs text-gray-400 uppercase">Entry</p>
                      <p className="font-mono font-bold text-white">{signal.entry}</p>
                    </div>
                    <div className="bg-gray-900/50 p-3 rounded-lg">
                      <p className="text-xs text-gray-400 uppercase">Stop Loss</p>
                      <p className="font-mono font-bold text-red-400">{signal.sl}</p>
                    </div>
                    <div className="bg-gray-900/50 p-3 rounded-lg">
                      <p className="text-xs text-gray-400 uppercase">Take Profit</p>
                      <p className="font-mono font-bold text-green-400">{signal.tp}</p>
                    </div>
                  </div>

                  {/* Right: Action */}
                  <div className="flex flex-col items-end gap-2 min-w-[140px]">
                    <div className="flex items-center gap-2">
                      <span className="text-sm text-gray-400">Confidence:</span>
                      <span className={`font-bold ${
                        signal.confidence > 90 ? 'text-green-400' : 
                        signal.confidence > 80 ? 'text-yellow-400' : 'text-orange-400'
                      }`}>
                        {signal.confidence}%
                      </span>
                    </div>
                    
                    <button
                      onClick={() => executeSignal(signal.pair)}
                      disabled={executed.includes(signal.pair)}
                      className={`w-full py-2 px-4 rounded-lg font-bold text-sm transition ${
                        executed.includes(signal.pair)
                          ? 'bg-green-600 text-white cursor-not-allowed'
                          : signal.type === 'BUY'
                            ? 'bg-green-600 hover:bg-green-700 text-white'
                            : 'bg-red-600 hover:bg-red-700 text-white'
                      }`}
                    >
                      {executed.includes(signal.pair) ? '✓ Executed' : `Execute ${signal.type}`}
                    </button>
                  </div>

                </div>
              </div>
            ))}
          </div>
        )}
      </div>
    </div>
  );
}
